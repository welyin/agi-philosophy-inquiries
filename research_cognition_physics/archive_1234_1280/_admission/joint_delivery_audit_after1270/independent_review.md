$$
\mathfrak I_a^{j,\kappa}(\rho_{CR})
=\Pi_a^F U\bigl(|G_{j,\kappa}\rangle\langle G_{j,\kappa}|
\otimes\rho_{CR}\otimes\rho_{FE}\bigr)U^\dagger\Pi_a^F.
$$
