$$
|\mathrm{Sp}(2q,2)|=\prod_{j=1}^{q}(4^j-1)2^{2j-1},

|\mathrm{Cliff}_q|=4^q|\mathrm{Sp}(2q,2)|.

$$
\frac{4^m2^n-1}{4^{m+n}-1}\le 2^{-n};
$$
