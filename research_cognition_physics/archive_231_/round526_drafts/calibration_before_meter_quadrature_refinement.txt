"""Round 526: a certified finite decoder for the same self-gated quantum model.

The unknown x changes the preparation only. Hamiltonian and decoder are fixed.
Integer interval arithmetic certifies the calibration; FFT propagation is a
separate finite-resolution diagnostic, not the proof of the confidence bound.
"""
import argparse
from fractions import Fraction as F
import hashlib
import json
import math
from pathlib import Path
import numpy as np
import self_gated_reference_limit as old

HERE = Path(__file__).resolve().parent
TARGET = HERE/'self_gated_reference_calibration_results.json'
T = F(1,16)
GAIN = F(1,3200)
DELTA = F(1,16)
NU = F(1,10)
SCALE = 1 << 64


def ceildiv(a,b):
    return -((-a)//b)


class Interval:
    """Outward-rounded intervals at a fixed binary scale; integer operations."""
    def __init__(self,lo,hi=None):
        self.lo, self.hi = int(lo), int(lo if hi is None else hi)
        assert self.lo <= self.hi

    @staticmethod
    def of(value):
        if isinstance(value,Interval):
            return value
        value = F(value)*SCALE
        return Interval(value.numerator//value.denominator,
                        ceildiv(value.numerator,value.denominator))

    def __add__(self,other):
        other = self.of(other)
        return Interval(self.lo+other.lo,self.hi+other.hi)

    __radd__ = __add__

    def __neg__(self):
        return Interval(-self.hi,-self.lo)

    def __sub__(self,other):
        return self+-self.of(other)

    def __rsub__(self,other):
        return self.of(other)+-self

    def __mul__(self,other):
        other = self.of(other)
        ends = [a*b for a in (self.lo,self.hi) for b in (other.lo,other.hi)]
        return Interval(min(ends)//SCALE,ceildiv(max(ends),SCALE))

    __rmul__ = __mul__

    def __truediv__(self,other):
        other = self.of(other)
        assert other.lo > 0
        ends = [F(a*SCALE,b) for a in (self.lo,self.hi) for b in (other.lo,other.hi)]
        lower,upper = min(ends),max(ends)
        return Interval(lower.numerator//lower.denominator,
                        ceildiv(upper.numerator,upper.denominator))


def exp_interval(x):
    x = Interval.of(x)
    assert -SCALE <= x.lo <= x.hi <= SCALE
    term = total = Interval.of(1)
    for k in range(1,25):
        term = term*x/k
        total = total+term
    # |u|<=1: Taylor remainder <= e/25! < 3/25!.
    rem = ceildiv(3*SCALE,math.factorial(25))
    return Interval(total.lo-rem,total.hi+rem)


_sqrt5 = math.isqrt(5*SCALE*SCALE)
SQRT5 = Interval(_sqrt5,_sqrt5+1)
IVEC = [-Interval.of(1)/SQRT5, Interval.of(2)/SQRT5,Interval.of(-1)]


def gradient_interval(q):
    f = Interval.of(1)/(1+exp_interval(-2*(q[0]-1)))
    fp = 2*f*(1-f)
    s = sum(v*a for v,a in zip(IVEC,q))
    out = [f*s*v for v in IVEC]
    out[0] = out[0]+fp*s*s/2
    out[1] = out[1]+F(5,4)*q[1]
    return out


def calibration_entry(x,panels=64):
    q0 = [Interval.of(x),Interval.of(0),Interval.of(0)]
    a0 = gradient_interval(q0)
    value = Interval.of(0)
    for k in range(panels):
        s = T*F(2*k+1,2*panels)
        q1 = [q-a*s*s/2 for q,a in zip(q0,a0)]
        value = value-(T-s)*gradient_interval(q1)[2]*(T/panels)
    # Final dyadic table entries are finite integer data, not exact-real oracles.
    tick = round(F(value.lo+value.hi,2*SCALE)*(1<<32))
    center = F(tick,1<<32)
    rounding = max(abs(center-F(value.lo,SCALE)),abs(center-F(value.hi,SCALE)))
    return tick,rounding,[value.lo,value.hi]


def certificate():
    g,h,d = F(9),F(10),F(13)
    r = h*T*T
    tail = (g/h)*r**3/(math.factorial(6)*(1-r/56))
    second = 3*h*g*T+d*g*g*T**3
    midpoint = second*T**3/(24*64**2)
    source = [F(3,4)+F(j,128) for j in range(65)]
    entries = [calibration_entry(x) for x in source]
    table = [row[0] for row in entries]
    arithmetic = max(row[1] for row in entries)
    error = tail+midpoint+arithmetic
    assert all(table[j]>table[j+1] for j in range(64))
    assert 2*error < GAIN/128
    inverse_error = F(1,128)+error/GAIN+F(1,512)
    assert inverse_error < DELTA/4
    # Fixed finite input record alphabet, including clipped tails.
    limit = 3*T*T/2
    step = GAIN*DELTA/2
    count = 1+int(2*limit/step)
    assert count==1201 and F(count-1)*step==2*limit
    decoder = []
    for j in range(count):
        w = -limit+j*step
        # Exact rational interpolation of the finite dyadic calibration table.
        if w >= F(table[0],1<<32):
            estimate = source[0]
        elif w <= F(table[-1],1<<32):
            estimate = source[-1]
        else:
            k = next(k for k in range(64) if F(table[k+1],1<<32)<=w<=F(table[k],1<<32))
            left,right = F(table[k],1<<32),F(table[k+1],1<<32)
            estimate = source[k]+(left-w)/(left-right)/128
        decoder.append(round(estimate*256))
    b = 1<<21
    probability = F(19,b)+4*(2+NU*NU)/(b*b*GAIN*GAIN*DELTA*DELTA)
    assert probability < F(1,200)
    # The conservative Jacobian and wavepacket bounds are rational statements.
    assert F(93,128)**2/5 > F(8,25)**2
    assert F(4,25)-F(317,24)*T*T >= F(2,25)
    assert F(13*29+7*123+577,96) < 19
    def packed(x):
        return dict(fraction=str(x),decimal=float(x))
    return dict(time=packed(T),unknown_interval=[.75,1.25],gain_lower=packed(GAIN),
        error_tolerance=packed(DELTA),amplitude_b=b,
        gaussian_variance_upper=2.,meter_variance=float(NU*NU),
        state_distance_upper=packed(F(19,b)),failure_upper=packed(probability),
        success_lower=packed(1-probability),picard_tail=packed(tail),
        midpoint_error=packed(midpoint),arithmetic_and_table_rounding=packed(arithmetic),
        calibration_error=packed(error),inverse_table_error=packed(inverse_error),
        symbols=count,bits=math.ceil(math.log2(count)),record_limit=packed(limit),
        record_step=packed(step),source_table_step=1/128,
        calibration_ticks_2_power_minus_32=table,
        calibration_quadrature_intervals_at_2_power_64=[row[2] for row in entries],
        decoder_ticks_2_power_minus_8=decoder,
        input_energy_upper=packed(F(5,32)*b*b+F(25,16)))


def orbit(x,end=float(T),steps=32):
    dt = end/steps
    state = np.r_[float(x)*old.E,np.zeros(3),np.eye(6).ravel()]
    rows = [state.copy()]
    for _ in range(steps):
        k1=old.rhs(state,1.,False)
        k2=old.rhs(state+dt*k1/2,1.,False)
        k3=old.rhs(state+dt*k2/2,1.,False)
        k4=old.rhs(state+dt*k3,1.,False)
        state += dt*(k1+2*k2+2*k3+k4)/6
        rows.append(state.copy())
    return np.array(rows)


def cdf(x):
    return np.array([.5*(1+math.erf(float(y)/math.sqrt(2))) for y in np.ravel(x)]).reshape(x.shape)


def quantum(x,b,cert,size=40,steps=24):
    end=float(T)
    grid=np.linspace(-5,5,size,endpoint=False)
    dx=float(grid[1]-grid[0])
    eta=np.array(np.meshgrid(grid,grid,grid,indexing='ij'))
    wave=np.exp(-np.sum(eta*eta,axis=0)/2).astype(complex)
    wave/=math.sqrt(float(np.sum(abs(wave)**2))*dx**3)
    k=2*math.pi*np.fft.fftfreq(size,dx)
    kinetic=np.exp(-.5j*end/steps*(k[:,None,None]**2+k[None,:,None]**2+k[None,None,:]**2))
    path=orbit(x,end,2*steps)
    for n in range(steps):
        q=path[2*n+1,:3]
        shifted=q[:,None,None,None]+eta/b
        residual=b*b*(old.potential(shifted)-old.potential(q)
                 -np.einsum('i,ijkl->jkl',old.gradient(q),eta)/b)
        phase=np.exp(-.5j*end/steps*residual)
        wave*=phase
        wave=np.fft.ifftn(kinetic*np.fft.fftn(wave))
        wave*=phase
    q=path[-1,:3]
    s=path[-1,6:].reshape(6,6)
    cov=s@s.T/2
    width=(s[3:,:3]+1j*s[3:,3:])@np.linalg.inv(s[:3,:3]+1j*s[:3,3:])
    gauss=np.exp(.5j*np.einsum('ij,iklm,jklm->klm',width,eta,eta))
    gauss/=math.sqrt(float(np.sum(abs(gauss)**2))*dx**3)
    distance=math.sqrt(max(0.,1-min(1.,abs(np.vdot(gauss,wave)*dx**3))**2))
    marginal=np.sum(abs(wave)**2,axis=(0,1))*dx**3
    count=cert['symbols']; limit=cert['record_limit']['decimal']; h=cert['record_step']['decimal']
    boundaries=np.r_[-np.inf,-limit+h*(np.arange(count-1)+.5),np.inf]
    # Conditional meter CDF given each quantum position grid value.
    cumulative=cdf((b*(boundaries[None,:]-q[2])-grid[:,None])/float(NU))
    probability=marginal@np.diff(cumulative,axis=1)
    decoded=np.array(cert['decoder_ticks_2_power_minus_8'])/256
    success=float(np.sum(probability[abs(decoded-x)<=float(DELTA)]))
    exact_gauss=cdf(b*(boundaries-q[2])/math.sqrt(cov[2,2]+float(NU*NU)))
    gaussian_record=np.diff(exact_gauss)
    record_tv=float(np.sum(abs(probability-gaussian_record)))/2
    edge=float(np.sum(abs(wave[np.max(abs(eta),axis=0)>4])**2))*dx**3
    assert abs(float(marginal.sum())-1)<2e-12 and edge<1e-6
    assert distance <= min(1.,19/b)+1e-7
    return dict(x=x,b=b,grid_size=size,time_steps=steps,
        classical_probe_mean=float(q[2]),quantum_normalized_mean=float(q[2]+marginal@grid/b),
        state_half_trace_to_gaussian=distance,record_tv_to_gaussian=record_tv,
        finite_decoder_success=success,gaussian_decoder_success=float(gaussian_record[abs(decoded-x)<=float(DELTA)].sum()),
        hessian_probe_variance=float(cov[2,2]),norm_error=abs(float(marginal.sum())-1),edge_mass=edge)


def bisect_root(target,left,right):
    fl=orbit(left)[-1,2]-target
    for _ in range(48):
        mid=(left+right)/2
        fm=orbit(mid)[-1,2]-target
        if fl*fm<=0:
            right=mid
        else:
            left,fl=mid,fm
    return (left+right)/2


def run():
    cert=certificate()
    calibration=[]
    for j,x in enumerate(np.linspace(.75,1.25,65)):
        row=orbit(x)[-1]
        value=cert['calibration_ticks_2_power_minus_32'][j]/(1<<32)
        discrepancy=abs(row[2]-value)
        assert discrepancy < cert['calibration_error']['decimal']
        jacobian=float(row[6:].reshape(6,6)[2,0])
        assert jacobian <= -float(GAIN)
        calibration.append(dict(x=float(x),probe=float(row[2]),derivative=jacobian,table_error=float(discrepancy)))
    refined=float(np.max(abs(orbit(1.,steps=64)[-1]-orbit(1.,steps=32)[-1])))
    assert refined<1e-10
    samples=[quantum(x,b,cert) for x in (.75,1.,1.25) for b in (64.,512.,4096.)]
    extra=quantum(1.,512.,cert,48,48)
    base=samples[4]
    delta=abs(extra['finite_decoder_success']-base['finite_decoder_success'])
    # The physical meter convolution is narrower than coarse grid bins at small b;
    # test probabilities under independent refinement rather than asserting exact TV.
    assert delta<2e-3
    assert abs(extra['quantum_normalized_mean']-base['quantum_normalized_mean'])<1e-8
    peak=float(orbit(-1.)[-1,2])
    assert peak>0
    pair=[bisect_root(peak/2,-8.,-1.),bisect_root(peak/2,-1.,0.)]
    means=[]; variances=[]
    for x in pair:
        row=orbit(x)[-1]; s=row[6:].reshape(6,6)
        means.append(float(row[2])); variances.append(float((s@s.T/2)[2,2]+float(NU*NU)))
    assert abs(means[0]-means[1])<1e-14 and pair[1]-pair[0]>1
    v,w=sorted(variances)
    crossing=math.sqrt(v*w*math.log(w/v)/(w-v)) if w>v else 0.
    tv=2*(.5*(1+math.erf(crossing/math.sqrt(2*v)))-.5*(1+math.erf(crossing/math.sqrt(2*w))))
    return dict(date='2026-09-30',round=526,scientific_base_through_round=525,
        tests_run=8,failures=0,errors=0,
        certificates=cert,diagnostics=dict(calibration_grid=calibration,
        classical_step_refinement=refined,actual_quantum_records=samples,
        refined_quantum_record=extra,decoder_success_refinement=delta,
        same_center_pair=pair,same_center_probe_means=means,limiting_record_variances=variances,
        limiting_gaussian_tv=tv,limiting_two_point_minimax_failure_lower=(1-tv)/2),
        dependency_hashes={n:hashlib.sha256((HERE/n).read_bytes()).hexdigest() for n in (
            'self_gated_reference_limit.py','research_note_525.md','research_note_480.md','local_reference_probe_review.md')},
        scope=dict(same_hamiltonian_for_every_unknown_x=True,
        scalar_reference_amplitude_not_position=True,uniform_compact_interval_identifiability=True,
        actual_quantum_to_gaussian_explicit_bound=True,finite_certified_calibration_table=True,
        single_shot_finite_record_confidence=True,global_scalar_readout_consistency_excluded=True,
        all_input_diamond_bound=False,all_measurement_protocols_excluded=False,
        autonomous_terminal_reader_derived=False,three_dimensional_space_derived=False,
        continuum_nonlinear_field_quantization_completed=False,einstein_feedback_completed=False,
        stage_complete=False))
